<script src="https://cdn.jsdelivr.net/npm/chart.js"></script>
<script>
document.addEventListener('DOMContentLoaded', function() {
    // Loan Status Chart (Doughnut) - Mejorado con gradientes y animaciones
    const ctxStatus = document.getElementById('loanStatusChart');
    if (ctxStatus) {
        const ctx = ctxStatus.getContext('2d');
        
        // Crear gradientes vibrantes
        const activeGradient = ctx.createLinearGradient(0, 0, 0, 300);
        activeGradient.addColorStop(0, '#fbbf24');
        activeGradient.addColorStop(1, '#f59e0b');
        
        const paidGradient = ctx.createLinearGradient(0, 0, 0, 300);
        paidGradient.addColorStop(0, '#34d399');
        paidGradient.addColorStop(1, '#10b981');
        
        new Chart(ctx, {
            type: 'doughnut',
            data: {
                labels: ['Activos', 'Pagados'],
                datasets: [{
                    data: [<?= $active_loans ?>, <?= $paid_loans ?>],
                    backgroundColor: [activeGradient, paidGradient],
                    borderWidth: 0,
                    hoverOffset: 20,
                    borderRadius: 10,
                    spacing: 2
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: true,
                cutout: '75%',
                plugins: {
                    legend: {
                        position: 'bottom',
                        labels: {
                            padding: 20,
                            font: {
                                size: 14,
                                family: "'Inter', -apple-system, sans-serif",
                                weight: '600'
                            },
                            usePointStyle: true,
                            pointStyle: 'circle',
                            color: '#334155'
                        }
                    },
                    tooltip: {
                        backgroundColor: 'rgba(15, 23, 42, 0.95)',
                        padding: 16,
                        cornerRadius: 12,
                        titleFont: {
                            size: 15,
                            weight: 'bold'
                        },
                        bodyFont: {
                            size: 14
                        },
                        borderColor: 'rgba(255, 255, 255, 0.1)',
                        borderWidth: 1,
                        callbacks: {
                            label: function(context) {
                                const total = context.dataset.data.reduce((a, b) => a + b, 0);
                                const percentage = ((context.parsed / total) * 100).toFixed(1);
                                return ' ' + context.label + ': ' + context.parsed + ' (' + percentage + '%)';
                            }
                        }
                    }
                },
                animation: {
                    animateRotate: true,
                    animateScale: true,
                    duration: 2000,
                    easing: 'easeInOutCubic'
                }
            }
        });
    }

    // Income Chart (Bar) - Mejorado con gradientes y efectos
    const ctxIncome = document.getElementById('incomeChart');
    if (ctxIncome) {
        const ctx = ctxIncome.getContext('2d');
        
        // Crear gradiente para las barras
        const barGradient = ctx.createLinearGradient(0, 0, 0, 300);
        barGradient.addColorStop(0, '#818cf8');
        barGradient.addColorStop(0.5, '#6366f1');
        barGradient.addColorStop(1, '#4f46e5');
        
        new Chart(ctx, {
            type: 'bar',
            data: {
                labels: <?= json_encode($months) ?>,
                datasets: [{
                    label: 'Ingresos',
                    data: <?= json_encode($incomes) ?>,
                    backgroundColor: barGradient,
                    borderRadius: 12,
                    borderSkipped: false,
                    hoverBackgroundColor: '#4338ca',
                    barThickness: 'flex',
                    maxBarThickness: 60
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: true,
                plugins: {
                    legend: {
                        display: false
                    },
                    tooltip: {
                        backgroundColor: 'rgba(15, 23, 42, 0.95)',
                        padding: 16,
                        cornerRadius: 12,
                        titleFont: {
                            size: 15,
                            weight: 'bold'
                        },
                        bodyFont: {
                            size: 14
                        },
                        borderColor: 'rgba(255, 255, 255, 0.1)',
                        borderWidth: 1,
                        callbacks: {
                            label: function(context) {
                                return ' Ingresos: $' + context.parsed.y.toLocaleString('es-MX', {
                                    minimumFractionDigits: 2,
                                    maximumFractionDigits: 2
                                });
                            }
                        }
                    }
                },
                scales: {
                    y: {
                        beginAtZero: true,
                        grid: {
                            color: 'rgba(148, 163, 184, 0.1)',
                            drawBorder: false,
                            lineWidth: 1
                        },
                        border: {
                            display: false
                        },
                        ticks: {
                            font: {
                                size: 12,
                                family: "'Inter', -apple-system, sans-serif",
                                weight: '500'
                            },
                            color: '#64748b',
                            callback: function(value) {
                                return '$' + value.toLocaleString('es-MX');
                            },
                            padding: 12
                        }
                    },
                    x: {
                        grid: {
                            display: false,
                            drawBorder: false
                        },
                        border: {
                            display: false
                        },
                        ticks: {
                            font: {
                                size: 12,
                                family: "'Inter', -apple-system, sans-serif",
                                weight: '500'
                            },
                            color: '#64748b',
                            padding: 12
                        }
                    }
                },
                animation: {
                    duration: 2000,
                    easing: 'easeInOutCubic',
                    delay: (context) => {
                        let delay = 0;
                        if (context.type === 'data' && context.mode === 'default') {
                            delay = context.dataIndex * 100;
                        }
                        return delay;
                    }
                },
                interaction: {
                    intersect: false,
                    mode: 'index'
                }
            }
        });
    }
});
</script>
</body>
</html>
